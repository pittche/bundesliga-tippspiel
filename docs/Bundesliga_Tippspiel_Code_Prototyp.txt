    <script>
        // --- State Management ---
        const appState = {
            currentTab: 'simulation',
            simulation: {
                baseStrengthA: 60, // Fixed statistical strength
                baseStrengthB: 40,
                trainerA: 90,      // Slider value
                moodA: 80,         // Slider value
                trainerB: 85,
                // Resulting probabilities
                probA: 55,
                probDraw: 25,
                probB: 20
            }
        };

        // --- Chart Instances ---
        let predictionChartInstance = null;
        let trendChartInstance = null;
        let radarChartInstance = null;

        // --- Navigation Logic ---
        function switchTab(tabId) {
            // Update State
            appState.currentTab = tabId;

            // UI Updates: Hide all sections, remove active class
            ['simulation', 'datasources', 'trends', 'roadmap'].forEach(id => {
                document.getElementById(`${id}-view`).classList.add('hidden');
                document.getElementById(`tab-${id}`).classList.remove('active-tab');
                document.getElementById(`tab-${id}`).classList.remove('text-gray-900');
                document.getElementById(`tab-${id}`).classList.add('text-gray-500');
            });

            // Show selected section, add active class
            document.getElementById(`${tabId}-view`).classList.remove('hidden');
            const activeBtn = document.getElementById(`tab-${tabId}`);
            activeBtn.classList.add('active-tab', 'text-gray-900');
            activeBtn.classList.remove('text-gray-500');

            // Trigger Chart renders if needed (charts need visibility to render correctly)
            if (tabId === 'simulation' && predictionChartInstance) {
                predictionChartInstance.resize(); 
            }
            if (tabId === 'trends') {
                setTimeout(() => {
                    initTrendChart();
                    initRadarChart();
                }, 10); // Slight delay to ensure DOM is visible
            }
        }

        // --- Simulation Logic (The "Algorithm" Prototype) ---
        function calculateProbability() {
            // Weights extracted from user prompt logic
            const weightStats = 0.6;
            const weightSoft = 0.4; // Soft factors (Trainer, Mood, Rumors)

            // Normalize slider inputs (0-100) to factor (0.5 - 1.5 impact)
            // Low slider = Bad mood/Crisis = Low factor
            const factorTrainerA = 0.5 + (document.getElementById('input-trainer-a').value / 100);
            const factorMoodA = 0.5 + (document.getElementById('input-mood-a').value / 100);
            
            const factorTrainerB = 0.5 + (document.getElementById('input-trainer-b').value / 100);
            
            // Calculate "Effective Strength"
            // Base strength is modified by the average of soft factors
            const softScoreA = (factorTrainerA + factorMoodA) / 2;
            const softScoreB = factorTrainerB; // Simplified for demo

            const effectiveA = (appState.simulation.baseStrengthA * weightStats) + (appState.simulation.baseStrengthA * softScoreA * weightSoft);
            const effectiveB = (appState.simulation.baseStrengthB * weightStats) + (appState.simulation.baseStrengthB * softScoreB * weightSoft);

            const total = effectiveA + effectiveB;
            
            // Calculate Percentages (simplified draw logic)
            const rawProbA = (effectiveA / total) * 100;
            const rawProbB = (effectiveB / total) * 100;
            
            // Artificial Draw Probability based on closeness
            const diff = Math.abs(rawProbA - rawProbB);
            let drawProb = 25; 
            if(diff > 30) drawProb = 15; // Mismatch -> less draws
            if(diff < 10) drawProb = 30; // Close match -> more draws

            const remainder = 100 - drawProb;
            appState.simulation.probA = (rawProbA / (rawProbA + rawProbB)) * remainder;
            appState.simulation.probB = (rawProbB / (rawProbA + rawProbB)) * remainder;
            appState.simulation.probDraw = drawProb;

            updateUI();
        }

        function updateUI() {
            // Update Labels
            const trainerAVal = document.getElementById('input-trainer-a').value;
            document.getElementById('val-trainer-a').innerText = trainerAVal < 40 ? "Kritisch!" : (trainerAVal < 70 ? "Angespannt" : "Stabil");
            document.getElementById('val-trainer-a').className = trainerAVal < 40 ? "text-red-600 font-bold" : (trainerAVal < 70 ? "text-yellow-600 font-bold" : "text-emerald-600 font-bold");

            const moodAVal = document.getElementById('input-mood-a').value;
            document.getElementById('val-mood-a').innerText = moodAVal < 40 ? "Unruhe/Streit" : "Fokussiert";
             document.getElementById('val-mood-a').className = moodAVal < 40 ? "text-red-600 font-bold" : "text-emerald-600 font-bold";

            const trainerBVal = document.getElementById('input-trainer-b').value;
            document.getElementById('val-trainer-b').innerText = trainerBVal < 40 ? "Kritisch!" : "Stabil";
             document.getElementById('val-trainer-b').className = trainerBVal < 40 ? "text-red-600 font-bold" : "text-emerald-600 font-bold";

            // Update Text Prediction
            const textEl = document.getElementById('prediction-text');
            if (appState.simulation.probA > 50) textEl.innerText = "Heimsieg wahrscheinlich";
            else if (appState.simulation.probB > 50) textEl.innerText = "Auswärtssieg wahrscheinlich";
            else textEl.innerText = "Offenes Spiel / Remis-Gefahr";

            // Update Chart
            if (predictionChartInstance) {
                predictionChartInstance.data.datasets[0].data = [
                    appState.simulation.probA, 
                    appState.simulation.probDraw, 
                    appState.simulation.probB
                ];
                predictionChartInstance.update();
            }
        }

        function updateSimulation() {
            calculateProbability();
        }
